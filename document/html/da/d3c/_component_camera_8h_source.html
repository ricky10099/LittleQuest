81 void SetUpdateTiming(ProcTiming timing) {
82 if(timing == ProcTiming::Update) {
83 camera_status_.on(CameraBit::UpdateOnUpdate);
84 camera_status_.off(CameraBit::UpdateOnPreUpdate);
85 camera_status_.off(CameraBit::UpdateOnPostUpdate);
