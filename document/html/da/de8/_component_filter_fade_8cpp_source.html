40void ComponentFilterFade::Init() {
41 __super::Init();
42
43 auto draw = [&]() {
44 // 現在のRenderTargetを保存
45 auto target_desc = GetRenderTarget();
46 auto render_target = target_desc.color_targets_[0];
47 //------------------------------------------------------
48 // ワークテクスチャにコピー
49 //------------------------------------------------------
50 CopyToRenderTarget(texture_work_.get(), render_target);
51
52 //------------------------------------------------------
53 // シェーダー適用コピーして描き戻す
54 //------------------------------------------------------
55 SetWriteZBufferFlag(false); // フィルター用にデプスバッファ更新を無効化
56
57 // 定数バッファ更新
58 auto* p = reinterpret_cast<Parameter*>(GetBufferShaderConstantBuffer(parameter_cb_));
59 {
60 p->resolution_ = float4(WINDOW_W, WINDOW_H, 1.0f / WINDOW_W, 1.0f / WINDOW_H);
61 p->alpha_ = fade_alpha_;
62
63 // モザイク幅
64 f32 mosaic_factor = 1.0f - fade_alpha_; // フェードアウトすると粗くなっていく
65 f32 mosaic_width = mosaic_width_ * mosaic_factor;
66
67 p->mosaic_width_ = std::max(1u, static_cast<u32>(mosaic_width));
68
69 UpdateShaderConstantBuffer(parameter_cb_); // シェーダー用定数バッファハンドルの定数バッファへの変更を適用する
70 }
71
72 // 定数バッファをb4に設定
73 SetShaderConstantBuffer(parameter_cb_, DX_SHADERTYPE_PIXEL, 4);
74
75 // 全画面描画
76 CopyToRenderTarget(render_target, texture_work_.get(), *shader_ps_);
77
78 SetShaderConstantBuffer(-1, DX_SHADERTYPE_PIXEL, 4); // b4
79 SetWriteZBufferFlag(true); // 元に戻す
80
81 //------------------------------------------------------
82 // 元に戻す
83 //------------------------------------------------------
84 SetRenderTarget(target_desc);
85 };
86
87 // 描画タイミングを設定して描画
88 SetProc("FilterDraw", draw, ProcTiming::Filter, ProcPriority::NORMAL);
89}

94void ComponentFilterFade::Update() {
95 f32 delta = GetDeltaTime();
96
97 if(finish_fade_)
98 return;
99
100 fade_alpha_ += fade_add_alpha_ * delta;
101 if(fade_alpha_ >= 1.0f) {
102 finish_fade_ = true;
103 fade_alpha_ = 1.0f;
104 }
105 if(fade_alpha_ < 0.0f) {
106 finish_fade_ = true;
107 fade_alpha_ = 0.0f;
108 }
109}
