52//! @brief スプリングアームの先からのマトリクス取得
53//! @return マトリクス
54matrix ComponentAttachModel::GetPutOnMatrix() const {
55 float trans[3] = {attach_model_offset_.x, attach_model_offset_.y, attach_model_offset_.z};
56 float rot[3] = {attach_model_rotate_.x, attach_model_rotate_.y, attach_model_rotate_.z};
57 float scale[3] = {1.0f, 1.0f, 1.0f};
58 matrix rmat;
59 RecomposeMatrixFromComponents(trans, rot, scale, rmat.f32_128_0);
60
61 matrix mat = matrix::identity();
62
63 if(auto object = object_.lock()) {
64 mat = object->GetMatrix();
65
66 if(auto target_model = object->GetComponent<ComponentModel>()) {
67 auto model_mat = target_model->GetMatrix();
68 int no = target_model->GetNodeIndex(object_node_name_);
69 if(no >= 0) {
70 mat = target_model->GetNodeMatrix(no);
71 }
72 }
73 }
74 mat = mul(rmat, mat);
75
76 return mat;
77}
