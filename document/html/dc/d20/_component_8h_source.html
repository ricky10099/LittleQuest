63//! クラス内ヘッダー宣言
64//! publicで BP_COMPONENT_DECL(クラス名, u8"任意の解説文字列") で宣言します
65//!
66//! @code

73#define BP_COMPONENT_DECL(CLASS, ...) \
74 using Super = ComponentClass; /*! 親クラスの型 : 上のクラス階層で定義されているClassを使うことで親クラスを定義 */ \
75 using ComponentClass = CLASS; /*! 自クラスの型 : これ以降は親クラスではなく自クラスを指す */ \
76 \
77 /*! 型情報 */ \
78 static inline ClassComponentType<CLASS> Type = ClassComponentType<CLASS>(#CLASS, &Super::Type, __VA_ARGS__); \
79 \
80 /*! 型情報を取得 */ \
81 virtual const ComponentTypeInfo* typeInfo() const { \
82 return &Type; \
83 } \
84 void* createComponentPtr(const ObjectPtr& obj) { \
85 return Type.createComponentPtr(obj); \
86 }
