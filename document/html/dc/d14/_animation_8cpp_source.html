43bool Animation::load(const Animation::Desc* desc, size_t desc_count) {
44 is_valid_ = true; // 先に一旦trueにする
45
46 // パラメーター初期化
47 for(u32 i = 0; i < desc_count; ++i) {
48 const auto& x = desc[i];
49
50 //------------------------------------------------------
51 // アニメーションリソース作成
52 // 既に同名ファイルがある場合は共有
53 //------------------------------------------------------
54 const std::string& resource_path = x.file_path_;
55
56 auto it = resource_pool.find(resource_path);
57 if(it == resource_pool.end()) { // 新規登録
58 resource_pool[resource_path] = std::make_shared<ResourceAnimation>(resource_path);
59 }
60
61 // 共有
62 std::shared_ptr<ResourceAnimation>& resource = resource_pool[resource_path];
63
64 descs_.push_back(x);
65 mv1_handles_.push_back(MV1DuplicateModel(*resource));
66
67 if(resource->isValid() == false) {
68 is_valid_ = false; // 一度でもエラーの場合はfalse
69 }
70
71 // 名前逆引きテーブルに登録
72 name_table_[desc[i].name_] = i;
73 }
74
75 return is_valid_;
76}

78//---------------------------------------------------------------------------
79//! 更新
80//---------------------------------------------------------------------------
81void Animation::update(f32 dt) {
82 assert(model_ && "アニメーションにモデルが設定されていません");
83
84 // 再生していない場合
85 if(isPlaying() == false)
86 return;
87
88 // ポーズ中の場合
89 if(is_paused_)
90 return;
91
92 //----------------------------------------------------------
93 // アニメーションを進める
94 //----------------------------------------------------------
95 float blend_ratio_total = 0.0f;
96 for(u32 i = 0; i < std::size(contexts_); ++i) {
97 auto& c = contexts_[i];
98
99 if(c.animation_index_ == -1)
100 continue;
101
102 auto& desc = descs_[c.animation_index_];
103
104 // アニメーション再生時間を進める
105 c.play_time_ += dt * 30.0f * desc.animation_speed_;
106
107 if(c.is_loop_) {
108 // アニメーション再生時間がアニメーションの総時間を越えていたらループさせる
109 if(c.animation_total_time_ <= c.play_time_) {
110 c.play_time_ -= c.animation_total_time_;
111 }
112 } else {
113 // アニメーション再生時間がアニメーションの総時間を越えていたら停止
114 if(c.animation_total_time_ <= c.play_time_) {
115 c.play_time_ = c.animation_total_time_;
116 c.is_playing_ = false;
117 }
118 }
119
120 // 新しいアニメーション再生時間をセット
121 DxLib::MV1SetAttachAnimTime(model_handle_, c.animation_attach_index_, c.play_time_);
122
123 //----------------------------------------------------------
124 // アニメーションブレンド
125 //----------------------------------------------------------
126
127 // ブレンドを進める
128 c.blend_ratio_ += ((1.0f / blend_time_) * dt) * ((i == 0) ? +1.0f : -1.0f);
129 c.blend_ratio_ = std::clamp(c.blend_ratio_, 0.0f, 1.0f);
130
131 blend_ratio_total += c.blend_ratio_;
132 }
133
134 //----------------------------------------------------------
135 // アニメーションブレンド
136 //----------------------------------------------------------
137 blend_ratio_total = std::max(FLT_EPSILON, blend_ratio_total);
138
139 for(u32 i = 0; i < std::size(contexts_); ++i) {
140 auto& c = contexts_[i];
141
142 float ratio = c.blend_ratio_;
143
144 DxLib::MV1SetAttachAnimBlendRate(model_handle_, c.animation_attach_index_, ratio / blend_ratio_total);
145
146 // 補間完了後は補間元のアニメーションを停止
147 if(i > 0) {
148 if(c.blend_ratio_ == 0.0f) { // float値の==判定は0.0fのみ正確に判定できる
149 detachAnimation(i);
150 c.is_playing_ = false;
151 }
152 }
153 }
154}

159void Animation::bindModel(Model* model) {
160 if(model == model_)
161 return;
162
163 // 旧モデルを解除
164 auto* last = model_;
165 if(last) {
166 // [DxLib]モデルとアニメーションの関連付けを解除
167 for(u32 i = 0; i < CONTEXT_COUNT; ++i) {
168 detachAnimation(i);
169 }
170
171 model_ = nullptr;
172 model_handle_ = -1;
173
174 last->bindAnimation(nullptr);
175 }
176
177 // 新モデルに登録
178 model_ = model;
179 if(model_) {
180 model_handle_ = *model_;
181 model_->bindAnimation(this);
182 }
183}

185//---------------------------------------------------------------------------
186//! アニメーションを再生する
187//---------------------------------------------------------------------------
188bool Animation::play(std::string_view name, bool is_loop, f32 blend_time, f32 start_time) {
189 // ブレンドの速度
190 blend_time_ = std::max(FLT_EPSILON, blend_time); // 0.0fにならないように
191
192 //----------------------------------------------------------
193 // [DxLib] 設定されているアニメーションを一旦解除
194 //----------------------------------------------------------
195 for(u32 i = 0; i < CONTEXT_COUNT; ++i) {
196 detachAnimation(i);
197 }
198
199 // 補間のために現在のアニメーションを補間前として設定
200 // [0] = 現在のアニメーション
201 // [1以降] = 以前のアニメーション
202 for(s32 i = CONTEXT_COUNT - 2; i >= 0; --i) {
203 contexts_[i + 1] = contexts_[i];
204 }
205
206 //----------------------------------------------------------
207 // 新規再生
208 //----------------------------------------------------------
209 {
210 // 名前からアニメーションを検索
211 std::string name_string(name);
212 auto it = name_table_.find(name_string);
213
214 if(it == name_table_.end()) {
215 return false; // 見つからなかった場合
216 }
217
218 auto& c = contexts_[0];
219
220 // アニメーション番号
221 c.animation_index_ = it->second;
222 // 現在の時間
223 c.play_time_ = start_time; // 開始位置は start_time から
224
225 c.is_playing_ = true;
226 c.is_loop_ = is_loop;
227 c.blend_ratio_ = 0.0f;
228 }
229
230 //----------------------------------------------------------
231 // アニメーション再生を設定
232 //----------------------------------------------------------
233 for(u32 i = 0; i < CONTEXT_COUNT; ++i) {
234 attachAnimation(i);
235 }
236
237 // ポーズ解除
238 is_paused_ = false;
239
240 // 強制的に再更新(時間は進めない)
241 // これを実行することで1フレームだけ不定値になってT-poseになる現象を回避
242 update(0.0f);
243
244 return true;
245}

316bool Animation::attachAnimation(s32 context_index) {
317 auto& c = contexts_[context_index];
318
319 if(c.animation_index_ == -1)
320 return false;
321
322 auto& desc = descs_[c.animation_index_];
323
324 // モデルにアニメーションを設定
325 auto animation_index = desc.animation_index_; // アニメーション番号
326 auto mv1_handle = mv1_handles_[c.animation_index_]; // [DxLib] アニメーションのMV1
327
328 c.animation_attach_index_ = DxLib::MV1AttachAnim(model_handle_, animation_index, mv1_handle, true);
329
330 // アニメーション総再生時間を取得
331 c.animation_total_time_ = DxLib::MV1GetAttachAnimTotalTime(model_handle_, c.animation_attach_index_);
332
333 // アニメーション再生時間を初期化
334 DxLib::MV1SetAttachAnimTime(model_handle_, c.animation_attach_index_, c.play_time_);
335
336 return true;
337}

356 // MV1モデルアニメーションの読み込み

365
366 // ハンドルの非同期読み込み処理が完了したら呼ばれる関数
367 auto finish_callback = []([[maybe_unused]] int mv1_handle, void* data) {
368 auto* resource = reinterpret_cast<ResourceAnimation*>(data);
369
370 // アクティブフラグを設定
371 resource->active_ = true;
372 };
373
374 //----------------------------------------------------------
375 // 非同期読み込み
376 //----------------------------------------------------------
377 SetUseASyncLoadFlag(true);
378 {
379 // モデルの読み込み
380 mv1_handle_ = DxLib::MV1LoadModel(resource_path.c_str());
381 DxLib::SetASyncLoadFinishCallback(mv1_handle_, (void (*)(int, void*))finish_callback, this);
382 }
383 SetUseASyncLoadFlag(false);
384}
