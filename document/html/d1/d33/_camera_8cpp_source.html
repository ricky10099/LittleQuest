128void Camera::ShakeCamera() {
129 if(!m_isShake) {
130 return;
131 }
132
133 if(shakeTimer > 0.0f) {
134 shakeTimer -= GetDeltaTime60(); // Assuming GetDeltaTime() gives the time elapsed since the last frame
135
136 // Calculate random offset for shake
137 float offsetX = (static_cast<float>(rand()) / RAND_MAX - 0.5f) * shakeMagnitude;
138 float offsetY = (static_cast<float>(rand()) / RAND_MAX - 0.5f) * shakeMagnitude;
139
140 // Apply offset to camera position
141 float3 shakePos = originalCameraPos + float3{offsetX, offsetY, 0};
142
143 m_pCamera.lock()->SetPosition(shakePos);
144
145 // Gradually decrease shake magnitude
146 shakeMagnitude *= 0.9f;
147
148 // Check if the shake duration has ended
149 if(shakeTimer <= 0.0f) {
150 // Reset the camera to its original position
151 m_pCamera.lock()->SetPosition(originalCameraPos);
152 m_isShake = false;
153 }
154 }
