23class DxLib_MV1MatrixCache {
24 public:
25 //! デフォルトコンストラクタ
26 DxLib_MV1MatrixCache() = default;
27
28 //! コンストラクタ
29 //! @param [in] MV1モデルハンドル
30 DxLib_MV1MatrixCache(int mv1_handle) {
31 initialize(mv1_handle);

34 //! 初期化
35 //! @param [in] MV1モデルハンドル
36 bool initialize(int mv1_handle) {
37 mv1_handle_ = mv1_handle;
38
39 // 配列を確保
40 auto triangle_list_count = DxLib::MV1GetTriangleListNum(mv1_handle);
41
42 for(s32 i = 0; i < 2; ++i) { // ダブルバッファの数
43 tlists_[i].resize(triangle_list_count);
44
45 for(s32 t = 0; t < triangle_list_count; t++) {
46 auto& tlist = tlists_[i][t];
47
48 //--------------------------------------------------------------
49 // 頂点座標保存用のStreamOutバッファを作成
50 //--------------------------------------------------------------
51 // [DxLib] 内部のDirect3Dデバイスを取得
52 auto* d3d_device = reinterpret_cast<ID3D11Device*>(const_cast<void*>(DxLib::GetUseDirect3D11Device()));
53
54 u32 vertex_count = MV1GetTriangleListVertexNum(mv1_handle, t);
55 vertex_count *= 3; // インデックスバッファを単純な三角形に展開した場合に最大3倍の頂点になる可能性
56 {
57 // DirectX11では構造化バッファを頂点バッファで利用できない
58 // 代わりにByteAddressBufferでバインドします。
59 D3D11_BUFFER_DESC desc{
60 .ByteWidth = sizeof(float4) * vertex_count, // バッファーのサイズ (バイト単位)
61 .Usage = D3D11_USAGE_DEFAULT, // メモリの配置場所 (読み取りと書き込み方法)
62 .BindFlags = D3D11_BIND_STREAM_OUTPUT | D3D11_BIND_SHADER_RESOURCE, // バインド用途
63 .CPUAccessFlags = 0, // CPUアクセス許可フラグ (殆どの場合0)
64 .MiscFlags = D3D11_RESOURCE_MISC_BUFFER_ALLOW_RAW_VIEWS, // ByteAddressBuffer
65 .StructureByteStride = 0, // 構造化バッファの構造体あたりの幅
66 };
67
68 HRESULT hr = d3d_device->CreateBuffer(&desc, nullptr, &tlist.d3d_position_buffer_);
69 if(hr != S_OK) {
70 return false;
71 }
72 }
73
74 // StreamOutバッファ用のShaderResourceViewを作成(ByteAddressBuffer)
75 {
76 Microsoft::WRL::ComPtr<ID3D11ShaderResourceView> d3d_srv;
77
78 D3D11_SHADER_RESOURCE_VIEW_DESC desc{};
79 desc.Format = DXGI_FORMAT_R32_TYPELESS; // float型 (TYPELESS指定が必要)
80 desc.ViewDimension = D3D11_SRV_DIMENSION_BUFFEREX; // BufferEx
81 desc.BufferEx.FirstElement = 0; // 最初の要素インデックス番号
82 desc.BufferEx.Flags = D3D11_BUFFEREX_SRV_FLAG_RAW; // ByteAddressBuffer
83 desc.BufferEx.NumElements = vertex_count * 4; // float * 4 * 頂点数
84
85 HRESULT hr = d3d_device->CreateShaderResourceView(tlist.d3d_position_buffer_.Get(), &desc, &d3d_srv);
86 if(hr != S_OK) {
87 return false;
88 }
89 tlist.d3d_srv_position_buffer_ = std::move(d3d_srv);
90 }
91 }
92 }
93 return true;
94 }
95
96 //! ダブルバッファを交換
97 void flip() {
98 std::swap(tlists_[0], tlists_[1]);
99 }
100
101 //! StreamOut用座標バッファを取得
102 ID3D11Buffer* position_buffer(u32 tlist_index) const {
103 // [0] = 現在のバッファ
104 return tlists_[0][tlist_index].d3d_position_buffer_.Get();
105 }
106
107 //! StreamOut用座標バッファを取得
108 ID3D11ShaderResourceView* srv(u32 tlist_index) const {
109 // [1] = 1フレーム前のバッファ
110 return tlists_[1][tlist_index].d3d_srv_position_buffer_.Get();
111 }
112
113 private:
114 struct Tlist {
115 Microsoft::WRL::ComPtr<ID3D11Buffer> d3d_position_buffer_;
116 Microsoft::WRL::ComPtr<ID3D11ShaderResourceView> d3d_srv_position_buffer_;
117 };
118
119 int mv1_handle_ = 0; //!< [DxLib] 対象のMV1モデルハンドル
120 std::vector<Tlist> tlists_[2]; //!< TListごとの情報
121};

126bool Model::load(std::string_view path) {
127 if(!textureIBL_diffuse_)
128 textureIBL_diffuse_ = std::make_shared<Texture>("data/IBL/iblDiffuseHDR.dds");
129 if(!textureIBL_specular_)
130 textureIBL_specular_ = std::make_shared<Texture>("data/IBL/iblSpecularHDR.dds");
131
132 //----------------------------------------------------------
133 // モデルリソース読み込み
134 //----------------------------------------------------------
135 {
136 std::string resource_path = std::string(path);
137
138 auto it = resource_model_pool.find(resource_path);
139 if(it == resource_model_pool.end()) { // 新規登録
140 resource_model_pool[resource_path] = std::make_shared<ResourceModel>(path);
141 }
142
143 // 共有
144 resource_model_ = resource_model_pool[resource_path];
145 }
146
147 // 以前ロードしたデータは削除する
148 if(animation_) {
149 animation_->bindModel(nullptr);
150 }
151
152 // [DxLib] MV1モデルを解放 (複製されたハンドル)
153 if(mv1_handle_ != -1) {
154 MV1DeleteModel(mv1_handle_);
155 mv1_handle_ = -1;
156 }
157
158 // 初期化を要求
159 need_initialize_ = true;
160
161 // 初回のみ読み込み
162 if(shader_vs_ == nullptr || shader_ps_ == nullptr) {
163 //----------------------------------------------------------
164 // シェーダーを読み込み
165 //----------------------------------------------------------
166 // 頂点シェーダー
167 // DX_MV1_VERTEX_TYPE_NUM 個のシェーダーバリエーションを生成
168 shader_vs_ = std::make_shared<ShaderVs>("data/Shader/vs_model", VS_VARIANT_COUNT);
169
170 // ピクセルシェーダー
171 shader_ps_ = std::make_shared<ShaderPs>("data/Shader/ps_model", PS_VARIANT_COUNT);
172
173 //----------------------------------------------------------
174 // デフォルトテクスチャを読み込み
175 //----------------------------------------------------------
176 tex_null_white_ = std::make_shared<Texture>("data/System/null_white.dds");
177 tex_null_black_ = std::make_shared<Texture>("data/System/null_black.dds");
178 tex_null_normal_ = std::make_shared<Texture>("data/System/null_normal.dds");
179 }
180
181 // 速度バッファ生成用のStreamOutジオメトリシェーダーを作成
182 {
183 shader_gs_streamout_position_ = std::make_shared<ShaderGs>("data/Shader/gs_model_streamout_position", 1);
184 shader_gs_composite_prev_position_ = std::make_shared<ShaderGs>("data/Shader/gs_model_composite_prev_position", 1);
185
186 // StreamOut用のバッファ出力レイアウト
187 // ストリーム番号, "出力要素の型", 出力要素のインデックス, 書き込みを開始するエントリのコンポーネント(0-1), コンポーネントの数(1-4), ストリーム出力バッファースロット番号(0-3)
188 D3D11_SO_DECLARATION_ENTRY so_declaration_entry[]{
189 {0, "POSITION", 0, 0, 4, 0},
190 };
191 std::array<u32, 1> strides = {sizeof(float) * 4};
192
193 // [DxLib] 内部のDirect3Dデバイスを取得
194 auto* d3d_device = reinterpret_cast<ID3D11Device*>(const_cast<void*>(DxLib::GetUseDirect3D11Device()));
195
196 auto [shader_binary, shader_binary_size] = shader_gs_streamout_position_->shader_bytecode();
197
198 HRESULT hr = d3d_device->CreateGeometryShaderWithStreamOutput(
199 shader_binary, // [in] シェーダーバイナリ
200 shader_binary_size, // [in] シェーダーバイナリのサイズ
201 so_declaration_entry, // [in] StreamOut用のバッファ出力レイアウト
202 static_cast<u32>(std::size(so_declaration_entry)), // [in] StreamOut用のバッファ出力レイアウト配列数
203 strides.data(), // [in] バッファーストライドの配列
204 static_cast<u32>(strides.size()), // [in] バッファーストライドの配列数
205 D3D11_SO_NO_RASTERIZED_STREAM, // [in] ラスタライザーステージに送信されるストリームのインデックス番号
206 nullptr, // [in] ID3D11ClassLinkageのポインタ
207 &d3d_shader_gs_streamout_position_); // [out] D3Dジオメトリシェーダー
208
209 if(hr != S_OK) {
210 return false;
211 }
212 }
213
214 return resource_model_->isValid();
215}

229void Model::render(ShaderVs* override_vs, ShaderPs* override_ps) {
230 if(!resource_model_)
231 return;
232
233 if(!resource_model_->isActive()) {
234 // ロードが終わっていない間は軽量モデルキャッシュ側を描画
235 auto* model_cache = resource_model_->modelCache();
236 model_cache->render(mat_world_);
237
238 return;
239 }
240
241 // ロード終了していたらハンドルを複製
242 on_initialize();
243
244 for(s32 frame = 0; frame < frameCount(); ++frame) {

246 }

249//---------------------------------------------------------------------------
250//! フレーム番号指定で描画
251//---------------------------------------------------------------------------
252void Model::renderByFrame(s32 frame_index, ShaderVs* override_vs, ShaderPs* override_ps) {
253 if(!resource_model_)
254 return;
255
256 if(!resource_model_->isActive())
257 return;
258
259 // ロード終了していたらハンドルを複製
260 on_initialize();
261
262 // フレーム番号が有効範囲外
263 if(MV1GetFrameNum(mv1_handle_) <= frame_index) {
264 return;
265 }
266
267 // ワールド行列を設定
268 MV1SetMatrix(mv1_handle_, mat_world_);
269
270 // シェーダーを使わない場合はDxLib関数を直接実行
271 if(!use_shader_) {
272 MV1DrawFrame(mv1_handle_, frame_index);
273 return;
274 }
275
276 //--------------------------------------------------
277 // テクスチャ設定の上書き
278 //--------------------------------------------------
279 bool override_normalmap = false;
280
281 if(overridedTextures_[static_cast<s32>(Model::TextureType::Diffuse)]) {
282 SetUseTextureToShader(0, *overridedTextures_[static_cast<s32>(Model::TextureType::Diffuse)]);
283 }
284 if(overridedTextures_[static_cast<s32>(Model::TextureType::Normal)]) {
285 SetUseTextureToShader(1, *overridedTextures_[static_cast<s32>(Model::TextureType::Normal)]);
286 override_normalmap = true; // 法線マップを使用
287 }
288 if(overridedTextures_[static_cast<s32>(Model::TextureType::Specular)]) {
289 SetUseTextureToShader(2, *overridedTextures_[static_cast<s32>(Model::TextureType::Specular)]);
290 }
291
292 //--------------------------------------------------
293 // シェーダーで描画
294 //--------------------------------------------------
295 ShaderVs* vs = override_vs ? override_vs : shader_vs_.get();
296 ShaderPs* ps = override_ps ? override_ps : shader_ps_.get();
297
298 // オリジナルシェーダーを使用をONにする
299 MV1SetUseOrigShader(true);
300
301 SetUseTextureToShader(11, *textureIBL_diffuse_);
302 SetUseTextureToShader(12, *textureIBL_specular_);
303
304 for(s32 mesh_index = 0; mesh_index < MV1GetFrameMeshNum(mv1_handle_, frame_index);
305 ++mesh_index) { // フレームに含まれるメッシュの数
306 s32 mesh = MV1GetFrameMesh(mv1_handle_, frame_index, mesh_index); // メッシュ番号
307
308 for(s32 tlist_index = 0; tlist_index < MV1GetMeshTListNum(mv1_handle_, mesh);
309 ++tlist_index) { // メッシュに含まれるトライアングルリストの数
310 auto tlist = MV1GetMeshTList(mv1_handle_, mesh, tlist_index); // トライアングルリスト番号
311
312 // トライアングルリストが使用しているマテリアルのインデックスを取得する
313 auto material_index = MV1GetTriangleListUseMaterial(mv1_handle_, tlist);
314
315 // 法線マップを使用しているかどうか
316 bool use_normalmap = MV1GetMaterialNormalMapTexture(mv1_handle_, material_index) != -1;
317
318 // 法線マップを使用しない場合はNull法線を登録しておく
319 if(!use_normalmap && !override_normalmap) {
320 SetUseTextureToShader(1, *tex_null_normal_);
321 }
322
323 //--------------------------------------------------
324 // シェーダーバリエーションを選択
325 //--------------------------------------------------
326 // 頂点データタイプ(DX_MV1_VERTEX_TYPE_1FRAME 等)
327 auto vertex_type = MV1GetTriangleListVertexType(mv1_handle_, tlist);
328 u32 variant_vs = vertex_type; // DXライブラリの頂点タイプをそのままバリエーション番号に
329
330 //--------------------------------------------------
331 // トライアングルリストを描画
332 //--------------------------------------------------
333 int handle_vs = vs->variant(variant_vs);
334 int handle_ps = *ps;
335
336 // シェーダーがない場合はオリジナルシェーダー利用を無効化
337 bool shader_enable = (handle_vs != -1) && (handle_ps != -1);
338 DxLib::MV1SetUseOrigShader(shader_enable);
339
340 // 頂点シェーダー
341 DxLib::SetUseVertexShader(handle_vs);
342
343 // ピクセルシェーダー
344 DxLib::SetUsePixelShader(handle_ps);
345
346 //--------------------------------------------------
347 // 速度バッファ生成のための2パス描画
348 //--------------------------------------------------
349 // StreamOutを利用して現在の頂点をバッファに出力保存
350 {
351 auto* d3d_context =
352 reinterpret_cast<ID3D11DeviceContext*>(const_cast<void*>(DxLib::GetUseDirect3D11DeviceContext()));
353
354 // ジオメトリシェーダーでStreamOutするための出力先を設定
355 Microsoft::WRL::ComPtr<ID3D11GeometryShader> old_shader_gs;
356 {
357 ID3D11Buffer* stream_out_buffer = mv1_matrix_cache_->position_buffer(tlist);
358
359 u32 offset[1] = {0};
360 d3d_context->SOSetTargets(1, &stream_out_buffer, offset);
361
362 d3d_context->GSGetShader(&old_shader_gs, nullptr, nullptr);
363 d3d_context->GSSetShader(d3d_shader_gs_streamout_position_.Get(), nullptr, 0);
364 }
365
366 // 描画 (StreamOut用)
367 MV1DrawTriangleList(mv1_handle_, tlist);
368
369 // StreamOutをもとに戻す
370 {
371 ID3D11Buffer* stream_out_buffer = nullptr;
372 u32 offset[1] = {0};
373 d3d_context->SOSetTargets(1, &stream_out_buffer, offset);
374
375 d3d_context->GSSetShader(old_shader_gs.Get(), nullptr, 0);
376 }
377 }
378
379 // (2) 1フレーム前の(1)を設定してGeometryShaderで頂点合成
380 {
381 auto* d3d_context =
382 reinterpret_cast<ID3D11DeviceContext*>(const_cast<void*>(DxLib::GetUseDirect3D11DeviceContext()));
383 u32 slot = 0;
384
385 // ジオメトリシェーダーで1フレーム前の頂点座標を合成
386 DxLib::SetUseGeometryShader(*shader_gs_composite_prev_position_);
387
388 // 1フレーム前のStreamOutされた頂点バッファをShaderResourceとして設定
389 Microsoft::WRL::ComPtr<ID3D11ShaderResourceView> old_srv;
390 {
391 ID3D11ShaderResourceView* srv = mv1_matrix_cache_->srv(tlist);
392 d3d_context->GSGetShaderResources(slot, 1, &old_srv);
393 d3d_context->GSSetShaderResources(slot, 1, &srv);
394 }
395
396 // 描画 (本来のシーン用)
397 MV1DrawTriangleList(mv1_handle_, tlist);
398
399 // もとに戻す
400 {
401 d3d_context->GSSetShaderResources(slot, 1, &old_srv);
402 DxLib::SetUseGeometryShader(-1);
403 }
404 }
405 }
406 }
407
408 // オリジナルシェーダー使用をOFFにする
409 DxLib::MV1SetUseOrigShader(false);
410
411 // 使い終わったらテクスチャ設定を解除
412 // 解除しておかないと他のモデル描画に影響あり。
413 DxLib::SetUseTextureToShader(0, -1);
414 DxLib::SetUseTextureToShader(1, -1);
415 DxLib::SetUseTextureToShader(2, -1);
416
417 DxLib::SetUseTextureToShader(11, -1);
418 DxLib::SetUseTextureToShader(12, -1);
419}
