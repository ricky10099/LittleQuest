47 public:
48 RigidBodyImpl(const JPH::ShapeRefC& shape, u16 layer, physics::MotionType motion_type) {
49 // ボディ自体の設定を作成します。ここで跳ね返り/摩擦係数のような他のプロパティも設定できます。
50 auto position = JPH::Vec3(0.0f, 0.0f, 0.0f);
51 auto rotation = JPH::Quat::sIdentity();
52
53 JPH::BodyCreationSettings body_settings(shape, // 形状
54 position, // 位置座標
55 rotation, // 回転姿勢
56 convertTo(motion_type), // 動的/静的/キネマティック
57 layer); // レイヤー番号
58
59 // CCD (Continuous Collision Detection) 有効
60 body_settings.mMotionQuality = JPH::EMotionQuality::LinearCast;
61
62 //------------------------------------------------------
63 // 剛体を作成
64 //------------------------------------------------------
65 // ボディを使い果たした場合はnullptrが返ります
66 jph_body_ = physics::Engine::bodyInterface()->CreateBody(body_settings);
67
68 if(jph_body_ == nullptr) {
69 assert(0 && "ボディ個数を最大数まで使い果たしました。");
70 return;
71 }
72
73 // ワールドに追加 (アクティブ化)
74 physics::Engine::bodyInterface()->AddBody(jph_body_->GetID(), JPH::EActivation::Activate);
75
76 // IDを保存
77 body_id_ = jph_body_->GetID();
78 }

425std::shared_ptr<physics::RigidBody> createRigidBody(const shape::ConvexHull& o, u16 layer, physics::MotionType motion_type,
426 f32 density) {
427 //----------------------------------------------------------
428 // 頂点配列を取得
429 //----------------------------------------------------------
430 JPH::Array<JPH::Vec3> vertexList;
431 JPH::IndexedTriangleList indexList;
432
433 for(auto& v: o.vertices_) {
434 JPH::Vec3 position{v.x * 0.01f, v.y * 0.01f, v.z * 0.01f};

436 }

438 //----------------------------------------------------------

453//---------------------------------------------------------------------------
454//! Mesh剛体を作成
455//---------------------------------------------------------------------------
456std::shared_ptr<physics::RigidBody> createRigidBody(const shape::Mesh& o, u16 layer) {
457 //----------------------------------------------------------
458 // 頂点配列を取得
459 //----------------------------------------------------------
460 JPH::VertexList vertexList;
461 JPH::IndexedTriangleList indexList;
462
463 for(auto& v: o.vertices_) {
464 JPH::Float3 position{v.x, v.y, v.z};
465 vertexList.emplace_back(std::move(position));
466 }
467 for(u32 i = 0; i < o.indices_.size(); i += 3) {
468 u32 i0 = o.indices_[i + 0];
469 u32 i1 = o.indices_[i + 1];
470 u32 i2 = o.indices_[i + 2];
471 JPH::IndexedTriangle triangle(i0, i1, i2, 0);
472 indexList.emplace_back(std::move(triangle));
473 }
474
475 JPH::PhysicsMaterialList materialList;
476 materialList.push_back(new JPH::PhysicsMaterialSimple("BPMeshMaterial", JPH::Color::sGetDistinctColor(0)));
477
478 //----------------------------------------------------------
479 // シェイプを生成
480 //----------------------------------------------------------
481 JPH::MeshShapeSettings shape_settings(vertexList, indexList, materialList);
482 JPH::ShapeSettings::ShapeResult result = shape_settings.Create();
483
484 if(result.HasError())
485 return nullptr;
486
487 // 作成
488 // メッシュ剛体は常に静的で生成されます
489 return std::make_shared<RigidBodyImpl>(result.Get(), layer, physics::MotionType::Static);
490}
