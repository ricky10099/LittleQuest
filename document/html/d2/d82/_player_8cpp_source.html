142void Player::GameAction() {
143 _movement = float3(0, 0, 0);
144 _hitTimer -= GetDeltaTime60();
145 _hitTimer = std::max(0.0f, _hitTimer);
146
147 if(_hitTimer <= 0 && _isHit) {
148 _isHit = false;
149 }
150
151 if(!_camera.lock()) {
152 _camera = Scene::GetObjectPtr<Camera>("PlayerCamera");
153 _cameraLength = _camera.lock()->GetComponent<ComponentSpringArm>()->GetSpringArmLength();
154 } else {
155 float3 v = _camera.lock()->CameraForward();
156 _selfMatrix = HelperLib::Math::CreateMatrixByFrontVector(v);
157 float3 dir = _camera.lock()->GetTranslate() - this->GetTranslate();
158 dir = normalize(dir);
159 _cameraCorrection.lock()->SetLine({0, 5, 0}, dir * _cameraLength);
160 }
161
162 _exSkillCamera.lock()->SetPositionAndTarget(_camera.lock()->GetTranslate(), GetTranslate());
163
164 InputHandle();
165
166 if(_currCombo != Combo::NO_COMBO && _playerState != PlayerState::ROLL) {
167 _playerState = PlayerState::ATTACK;
168 }
169
170 if(IsFloat3Zero(_movement) && _playerState != PlayerState::GET_HIT && _playerState != PlayerState::ATTACK &&
171 _playerState != PlayerState::ROLL && _playerState != PlayerState::DEAD) {
172 _playerState = PlayerState::IDLE;
173 }
174
175 switch(_playerState) {
176 case PlayerState::GET_HIT:
177 if(!_model.lock()->IsPlaying()) {
178 _playerState = PlayerState::IDLE;
179 }
180 break;
181 case PlayerState::ATTACK:
182 Attack();
183 break;
184 case PlayerState::ROLL:
185 _model.lock()->PlayAnimationNoSame(STR(PlayerState::ROLL));
186 _currCombo = Combo::NO_COMBO;
187 _weaponCollision.lock()->SetHitCollisionGroup((u32)ComponentCollision::CollisionGroup::NONE);
188 if(_model.lock()->IsPlaying()) {
189 _movement = {};
190 _isInvincible = true;
191 float3 vec = _model.lock()->GetMatrix().axisZ();
192 vec.y = 0;
193 _movement += -vec;
194 _movement *= 15.0f * GetDeltaTime60();
195 AddTranslate(_movement);
196 } else {
197 _isInvincible = false;
198 _playerState = PlayerState::IDLE;
199 }
200 break;
201 case PlayerState::WALK:
202 Walk();
203 break;
204 case PlayerState::DEAD:
205 Die();
206 break;
207 default:
208 Idle();
209 break;
210 }

241void Player::OnHit([[maybe_unused]] const ComponentCollision::HitInfo& hitInfo) {
242 if((u32)hitInfo.collision_->GetCollisionGroup() & (u32)ComponentCollision::CollisionGroup::WEAPON) {
243 auto* owner = hitInfo.hit_collision_->GetOwner();
244
245 Enemy* enemy;
246 if((enemy = dynamic_cast<Enemy*>(owner)) && _currCombo != Combo::NO_COMBO) {
247 bool inList = false;
248 for(int i = 0; i < _attackList.size(); i++) {
249 if(_attackList[i] == enemy->GetName().data()) {
250 inList = true;
251 break;
252 }
253 }
254
255 if(!inList) {
256 _attackList.push_back(enemy->GetName().data());
257 enemy->GetHit((int)(this->BASE_ATK * _comboList[_currCombo] * _componentCombo.lock()->ComboBuff()));
258 _isHit = true;
259 _hitTimer += HIT_PAUSE;
260 _componentCombo.lock()->AddCombo(_comboList[_currCombo]);
261 _playingEffect = PlayEffekseer3DEffect(_hitEffect);
262 PlaySoundMem(_swordHitSE, DX_PLAYTYPE_BACK);
263 ChangeVolumeSoundMem((int)(MAX_VOLUME * (Scene::GetSEVolume() / 100.0f)), _swordHitSE);
264 SetPosPlayingEffekseer3DEffect(_playingEffect, hitInfo.hit_position_.x, hitInfo.hit_position_.y,
265 hitInfo.hit_position_.z);
266 _camera.lock()->SetCameraShake(15, 5);
267 _camera.lock()->ShakeCamera();
268 }
269 }
270
271 BreakableObject* breakableObject;
272 if((breakableObject = dynamic_cast<BreakableObject*>(owner)) && _currCombo != Combo::NO_COMBO) {
273 bool inList = false;
274 for(int i = 0; i < _attackList.size(); i++) {
275 if(_attackList[i] == breakableObject->GetName().data()) {
276 inList = true;
277 break;
278 }
279 }
280
281 if(!inList) {
282 _attackList.push_back(breakableObject->GetName().data());
283 if(_currCombo == Combo::EXSKILL_1) {
284 breakableObject->GetHit(10);
285 } else {
286 breakableObject->GetHit();
287 }
288 _isHit = true;
289 _componentCombo.lock()->AddCombo(1);
290 _playingEffect = PlayEffekseer3DEffect(_hitBuildingEffect);
291 SetPosPlayingEffekseer3DEffect(_playingEffect, hitInfo.hit_position_.x, hitInfo.hit_position_.y,
292 hitInfo.hit_position_.z);
293 _camera.lock()->SetCameraShake(10, 5);
294 _camera.lock()->ShakeCamera();
295 }
296 }
297 }
298
299 if(_camera.lock()) {
300 if((u32)hitInfo.collision_->GetCollisionGroup() & (u32)ComponentCollision::CollisionGroup::ETC) {
301 if(hitInfo.hit_) {
302 _blockedDistance = GetDistance(this->GetTranslate(), hitInfo.hit_position_, true);
303 if(_blockedDistance < _cameraLength &&
304 hitInfo.hit_collision_->GetName() != _camera.lock()->GetAnchorCollisionName()) {
305 _cameraBlocked = true;
306 _blockedName = hitInfo.hit_collision_->GetOwner()->GetName().data();
307 _camera.lock()->SetCameraPositionAndTarget(
308 {_camera.lock()->GetCameraLocalPosition().x, _camera.lock()->GetCameraLocalPosition().y,
309 _cameraLength - _blockedDistance},
310 {0, 0, _cameraLength});
311 } else {
312 _cameraBlocked = false;
313 _camera.lock()->SetCameraPositionAndTarget(
314 {_camera.lock()->GetCameraLocalPosition().x, _camera.lock()->GetCameraLocalPosition().y, 0},
315 {0, 0, _cameraLength});
316 }
317 }
318 } else {
319 _camera.lock()->SetCameraPositionAndTarget(
320 {_camera.lock()->GetCameraLocalPosition().x, _camera.lock()->GetCameraLocalPosition().y, 0},
321 {0, 0, _cameraLength});
322 _cameraBlocked = false;
323 }
324 }
325
326 Super::OnHit(hitInfo);
327}

349void Player::InputHandle() {
350 DINPUT_JOYSTATE DInputState;
351 _cameraLength -= GetMouseWheelRotVol() * 3;
352 if(IsPadRepeat(PAD_ID::PAD_R)) {
353 ++_cameraLength;
354 }
355 if(IsPadRepeat(PAD_ID::PAD_L)) {
356 --_cameraLength;
357 }
358 _cameraLength = std::min((std::max(_cameraLength, 10.0f)), 100.0f);
359 _camera.lock()->SetCameraLength(_cameraLength);
360
361 if(_playerState != PlayerState::ROLL && _playerState != PlayerState::GET_HIT && _playerState != PlayerState::DEAD) {
362 _speedFactor = IsKeyRepeat(KEY_INPUT_LSHIFT) ? RUN_MULTIPLIER : 1.0f;
363
364 GetJoypadDirectInputState(DX_INPUT_PAD1, &DInputState);
365 switch(GetJoypadType(DX_INPUT_PAD1)) {
366 case DX_PADTYPE_DUAL_SENSE:
367 _movement += float3{-DInputState.X, 0, DInputState.Y};
368 _movement = mul(float4{_movement, 1.0f}, _selfMatrix).xyz;
369 break;
370 default:
371 break;
372 }
373
374 if(IsKeyRepeat(KEY_INPUT_W)) {
375 float3 vec = _selfMatrix.axisZ();
376 vec.y = 0;
377 _movement += -vec;
378 }
379 if(IsKeyRepeat(KEY_INPUT_D)) {
380 float3 vec = _selfMatrix.axisX();
381 vec.y = 0;
382 _movement += -vec;
383 }
384 if(IsKeyRepeat(KEY_INPUT_S)) {
385 float3 vec = _selfMatrix.axisZ();
386 vec.y = 0;
387 _movement += vec;
388 }
389 if(IsKeyRepeat(KEY_INPUT_A)) {
390 float3 vec = _selfMatrix.axisX();
391 vec.y = 0;
392 _movement += vec;
393 }
394
395 if(IsMouseDown(MOUSE_INPUT_LEFT) || IsPadOn(PAD_ID::PAD_A)) {
396 _playerState = PlayerState::ATTACK;
397
398 // もし攻撃してない場合はコンボ１に行く
399 // 攻撃しているかつチャージ攻撃ではないならコンボ中に判断する
400 if(_currCombo == Combo::NO_COMBO) {
401 _currCombo = Combo::NORMAL_COMBO1;
402 } else if(_currCombo != Combo::SPECIAL_ATTACK) {
403 _isCombo = true;
404 }
405 }
406
407 // 右クリック長押ししたらチャージ攻撃
408 if(IsMouseRepeat(MOUSE_INPUT_RIGHT, 1) || IsPadRepeat(PAD_ID::PAD_Y)) {
409 _chargeTime += GetDeltaTime60();
410 if(_chargeTime >= SPECIAL_CHARGE_TIME && !_charged) {
411 StopEffekseer3DEffect(_playingChargeEffect);
412 _charged = true;
413 }
414 // チャージ時間によってチャージエフェクトが変わる
415 if(IsEffekseer3DEffectPlaying(_playingChargeEffect) == -1) {
416 _playingChargeEffect = PlayEffekseer3DEffect(_pChargeList[(_chargeTime >= SPECIAL_CHARGE_TIME)]);
417 }
418 SetPosPlayingEffekseer3DEffect(_playingChargeEffect, GetTranslate().x, GetTranslate().y + 7, GetTranslate().z);
419 _speedFactor = 1.0;
420 } else if(_chargeTime > 0 && _currCombo == Combo::NO_COMBO) {
421 StopEffekseer3DEffect(_playingChargeEffect);
422 if(_chargeTime >= SPECIAL_CHARGE_TIME) {
423 _isCombo = true;
424 }
425 _charged = false;
426 _playerState = PlayerState::ATTACK;
427 _currCombo = Combo::SPECIAL_ATTACK;
428 _chargeTime = 0;
429 _speedFactor = RUN_MULTIPLIER;
430 }
431
432 if(_componentCombo.lock()->IsFullCharge() && IsKeyDown(KEY_INPUT_R)) {
433 _playerState = PlayerState::ATTACK;
434
435 if(_currCombo == Combo::NO_COMBO) {
436 _currCombo = Combo::EXSKILL_0;
437 }
438 }
439
440 if(!IsFloat3Zero(_movement)) {
441 _playerState = PlayerState::WALK;
442 }
443
444 // 必殺技途中は回避できない
445 if((IsKeyDown(KEY_INPUT_SPACE) || IsPadOn(PAD_ID::PAD_B)) && _currCombo != Combo::EXSKILL_0 &&
446 _currCombo != Combo::EXSKILL_1) {
447 _chargeTime = 0;
448 _charged = false;
449 StopEffekseer3DEffect(_playingChargeEffect);
450 _playerState = PlayerState::ROLL;
451 this->SetModelRotation();
452 }
453 }

551 // 必殺技演出
552 _exSkillTimer -= GetDeltaTime60();
553 _exSkillTimer = std::max(0.0f, _exSkillTimer);
554 float t = abs(1 - (_exSkillTimer / EXSKILL_PLAY_TIME));
555 float yawRad = (_model.lock()->GetRotationAxisXYZ().y) * DegToRad;
556 float distanceInFront = -5.0f;
557 float offsetX = sinf(yawRad) * distanceInFront;
558 float offsetZ = cosf(yawRad) * distanceInFront;
559 float offsetY = 8.0f;
560 float3 targetCamPos = {GetTranslate().x + offsetX, GetTranslate().y + offsetY, GetTranslate().z + offsetZ};
561 // 補間でアクションカメラの位置から正面に移る
562 float3 newCamPos = SlerpWithCenter(GetTranslate(), _camera.lock()->GetTranslate(), targetCamPos, t);
563 _exSkillCamera.lock()->SetCurrentCamera();
564 _exSkillCamera.lock()->SetPositionAndTarget(newCamPos, GetTranslate() + EXSKILL_CAMERA_TARGET_OFFSET);
565 if(!_model.lock()->IsPlaying()) {
566 _currCombo = Combo::EXSKILL_1;
567 _isPlayingEXSkill = false;
568 _camera.lock()->SetCurrentCamera();
569 _exSkillTimer = EXSKILL_PLAY_TIME;
570 }
571 if(!_playedFX) {
572 float3 fxRotation = {0, (_model.lock()->GetRotationAxisXYZ().y) * DegToRad, 0};
573 PlayAttackFX(_exSkillEffect, GetTranslate(), fxRotation, _exSkill0SE);
574 _playingEffect = SetSpeedPlayingEffekseer3DEffect(_playingEffect, 0.5f);
575 _playedFX = true;
576 }
577 break;
578 }
579 case Combo::EXSKILL_1:
580 AttackAnimation(STR(Combo::EXSKILL_1), _animList[STR(Combo::EXSKILL_1)], Combo::NO_COMBO, true);
581 _playingEffect = SetSpeedPlayingEffekseer3DEffect(_playingEffect, 2.0f);
582 if(!_playedFX) {
583 PlaySoundMem(_exSkill1SE, DX_PLAYTYPE_BACK);
584 ChangeVolumeSoundMem((int)(MAX_VOLUME * (Scene::GetSEVolume() / 100.0f)), _exSkill1SE);
585 _playedFX = true;
586 }
587 if(_model.lock()->GetPlayAnimationName() != STR(Combo::EXSKILL_1)) {
588 _isInvincible = false;
589 _componentCombo.lock()->ClearCombo();
590 }
591 break;
592 default:
593 _playerState = PlayerState::IDLE;
594 _currCombo = Combo::NO_COMBO;
595 break;
596 }
597}

599void Player::AttackAnimation(std::string animName, AnimInfo animInfo, Combo nextCombo, bool isEXSkill) {
600 if(_model.lock()->GetPlayAnimationName() != animName) {
601 _currAnimName = animName;
602 this->SetModelRotation();
603 _model.lock()->PlayAnimationNoSame(animName, false, 0.2f, _animList[animName].animStartTime);
604 _model.lock()->SetAnimationSpeed(animInfo.animStartSpeed);
605 _attackList.clear();
606 _playedFX = false;
607 }
608 _currAnimTime = _model.lock()->GetAnimationPlayTime();
609
610 // 指定のフレームに着いたら攻撃判定が有効にする
611 if(_currAnimTime > _animList[animName].triggerStartTime) {
612 // 敵に当たったらヒットストップが発生する
613 if(_isHit) {
614 _model.lock()->SetAnimationSpeed(animInfo.animSpeed * 0);
615 } else if(_slowMotion) {
616 _model.lock()->SetAnimationSpeed(animInfo.animSpeed * 0.01f);
617 } else {
618 _model.lock()->SetAnimationSpeed(animInfo.animSpeed);
619 }
620 if(isEXSkill) {
621 _exSkillCollision.lock()->SetHitCollisionGroup((u32)ComponentCollision::CollisionGroup::ENEMY |
622 (u32)ComponentCollision::CollisionGroup::ITEM);
623 _exSkillCollision.lock()->Overlap((u32)ComponentCollision::CollisionGroup::ENEMY |
624 (u32)ComponentCollision::CollisionGroup::ITEM);
625
626 } else {
627 _weaponCollision.lock()->SetHitCollisionGroup((u32)ComponentCollision::CollisionGroup::ENEMY |
628 (u32)ComponentCollision::CollisionGroup::ITEM);
629 }
630 }
631
632 // 指定のフレームに着いたら攻撃判定が無効にする
633 if(_currAnimTime > _animList[animName].triggerEndTime) {
634 _weaponCollision.lock()->SetHitCollisionGroup((u32)ComponentCollision::CollisionGroup::NONE);
635 _exSkillCollision.lock()->SetHitCollisionGroup((u32)ComponentCollision::CollisionGroup::NONE);
636 if(isEXSkill) {
637 _isInvincible = false;
638 _componentCombo.lock()->ClearCombo();
639 _exSkillCollision.lock()->Overlap((u32)ComponentCollision::CollisionGroup::ENEMY);
640 }
641 }
642
643 // 指定のフレームに着いたら入力次第アニメションが中断する
644 if(_currAnimTime > _animList[animName].animCutInTime) {
645 _currCombo = Combo::NO_COMBO;
646 if(_isCombo) {
647 _currCombo = nextCombo;
648 _isCombo = false;
649 }
650 }
651}
