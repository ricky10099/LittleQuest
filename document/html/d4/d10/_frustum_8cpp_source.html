25//---------------------------------------------------------------------------
26//! 更新
27//---------------------------------------------------------------------------
28void Frustum::update() {
29 //----------------------------------------------------------
30 // 行列を作成
31 //----------------------------------------------------------
32 // ビュー行列
33 mat_view_ = matrix::lookAtLH(position_, look_at_, world_up_);
34
35 // カメラのワールド行列はビュー行列の逆行列
36 mat_camera_world_ = inverse(mat_view_);
37
38 // 投影行列
39 switch(depth_mode_) {
40 case DepthMode::Default:
41 mat_proj_ = matrix::perspectiveFovLH(fovy_, aspect_ratio_, z_near_, z_far_);
42 break;
43 case DepthMode::Reverse:
44 // farとnearを入れ替えることでreverse-zになる
45 mat_proj_ = matrix::perspectiveFovLH(fovy_, aspect_ratio_, z_far_, z_near_);
46 break;
47 case DepthMode::ReverseInfinite:
48 // 無限遠まで高精度レンダリングする特殊投影行列
49 mat_proj_ = matrix::perspectiveFovInfiniteFarPlaneLH(fovy_, aspect_ratio_, z_near_);
50 break;
51 }
52
53 // 合成
54 mat_view_proj_ = mul(mat_view_, mat_proj_);
55}

60void Frustum::renderDebug() {
61 f32 zf = 10.0f; // 近クリップ距離
62 f32 zn = 1.0f; // 遠クリップ距離
63
64 // nearとfarの距離の矩形の幅・高さ
65 f32 nw = tanf(fovy_ * 0.5f) * zn * aspect_ratio_;
66 f32 nh = tanf(fovy_ * 0.5f) * zn;
67 f32 fw = tanf(fovy_ * 0.5f) * zf * aspect_ratio_;
68 f32 fh = tanf(fovy_ * 0.5f) * zf;
69
70 // カメラの姿勢情報
71 float3 axis_x = mat_camera_world_.axisX();
72 float3 axis_y = mat_camera_world_.axisY();
73 float3 axis_z = mat_camera_world_.axisZ();
74
75 //----------------------------------------------------------
76 // 前後四隅の点を計算
77 //----------------------------------------------------------
78 float3 v[8];
79 v[0] = position_ + axis_z * zn - axis_x * nw + axis_y * nh;
80 v[1] = position_ + axis_z * zn + axis_x * nw + axis_y * nh;
81 v[2] = position_ + axis_z * zn + axis_x * nw - axis_y * nh;
82 v[3] = position_ + axis_z * zn - axis_x * nw - axis_y * nh;
83 v[4] = position_ + axis_z * zf - axis_x * fw + axis_y * fh;
84 v[5] = position_ + axis_z * zf + axis_x * fw + axis_y * fh;
85 v[6] = position_ + axis_z * zf + axis_x * fw - axis_y * fh;
86 v[7] = position_ + axis_z * zf - axis_x * fw - axis_y * fh;
87
88 SetUseLighting(false);
89
90 //----------------------------------------------------------
91 // デプス無効でシルエットを描画
92 //----------------------------------------------------------
93 SetUseZBuffer3D(false);
94 {
95 DrawLine3D(cast(v[0]), cast(v[1]), GetColor(0, 0, 0));

97 DrawLine3D(cast(v[2]), cast(v[3]), GetColor(0, 0, 0));
98 DrawLine3D(cast(v[3]), cast(v[0]), GetColor(0, 0, 0));
99

102 DrawLine3D(cast(v[6]), cast(v[7]), GetColor(0, 0, 0));
103 DrawLine3D(cast(v[7]), cast(v[4]), GetColor(0, 0, 0));
104
105 DrawLine3D(cast(v[0]), cast(v[4]), GetColor(0, 0, 0));
106 DrawLine3D(cast(v[1]), cast(v[5]), GetColor(0, 0, 0));
107 DrawLine3D(cast(v[2]), cast(v[6]), GetColor(0, 0, 0));
108 DrawLine3D(cast(v[3]), cast(v[7]), GetColor(0, 0, 0));
109
110 DrawLine3D(cast(position_ + axis_z * zn), cast(position_ + axis_z * zf * 2.0f), GetColor(0, 0, 0));
111 }
112
113 for(u32 i = 0; i < 8; ++i) {
114 DrawSphere3D(cast(v[i]), 0.1f, 8, GetColor(255, 255, 0), 0, true);
115 }
116
117 //----------------------------------------------------------
118 // デプス有効でシルエットを描画
119 //----------------------------------------------------------
120 SetUseZBuffer3D(true);
121 {
122 DrawLine3D(cast(v[0]), cast(v[1]), GetColor(0, 255, 0));
123 DrawLine3D(cast(v[1]), cast(v[2]), GetColor(0, 255, 0));
124 DrawLine3D(cast(v[2]), cast(v[3]), GetColor(0, 255, 0));

130 DrawLine3D(cast(v[7]), cast(v[4]), GetColor(0, 255, 0));
131
132 DrawLine3D(cast(v[0]), cast(v[4]), GetColor(0, 255, 0));
133 DrawLine3D(cast(v[1]), cast(v[5]), GetColor(0, 255, 0));
134 DrawLine3D(cast(v[2]), cast(v[6]), GetColor(0, 255, 0));
135 DrawLine3D(cast(v[3]), cast(v[7]), GetColor(0, 255, 0));
136
137 DrawLine3D(cast(position_ + axis_z * zn), cast(position_ + axis_z * zf * 2.0f), GetColor(0, 255, 0));
138 }
139 SetUseLighting(true);
140}

270f32 Frustum::screenDistance(u32 resolution_height) const {
271 // +
272 // ／ |
273 // ／ |
274 // ／ | resolution_height /2
275 // ／ |
276 // ／）fovy_ / 2 |
277 //+-----------------+
278 // d スクリーンまでの距離
279 //
280 // tan(fovy_ / 2) = (resolution_height / 2) / d
281 //
282 return static_cast<f32>(resolution_height / 2) / tanf(fovy_ * 0.5f);
283}

302float3 Frustum::rayFromScreenPosition(float2 screen_position) {
303 // カメラの姿勢情報
304 float3 axis_x = mat_camera_world_.axisX();
305 float3 axis_y = mat_camera_world_.axisY();
306 float3 axis_z = mat_camera_world_.axisZ();
307
308 return screen_position.x * axis_x + screen_position.y / aspect_ratio_ * axis_y + axis_z;
309}
