652 }
653
654 if(IsEffekseer3DEffectPlaying(_playingEffect) == -1 && !_isAngry) {
655 _isAngry = true;
656 _playingEffect = PlayEffekseer3DEffect(_powerUpEffect);
657 SetPosPlayingEffekseer3DEffect(_playingEffect, GetTranslate().x, GetTranslate().y, GetTranslate().z);
658 }

663 _angryBox.reset();

665 }
666 }
667 if(_currAnimTime > _animList[STR(MawJLaygoAnim::ANGRY_AURA)].animCutInTime) {
668 _combo++;
669 _isHitPlayer = false;
670 _isHitItem = false;
671 }
