128void Stage01::Update() {
129 float t;
130 float3 newCamPos;
131 float3 newCamTarget;
132 float newFOV;
133
134#ifndef _DEBUG
135 HideMouse(true);
136#endif // !_DEBUG
137
138 switch(scene_state) {
139 case Scene::SceneState::TRANS_IN:
140 _player.lock()->SetHideHP(true);
141 _boss.lock()->SetHideHP(true);
142 for(auto& mob: _mob) {
143 mob.lock()->SetHideHP(true);
144 }
145
146 if(FadeIn()) {
147 _boss.lock()->PlayTaunt();
148 _playerCamera.lock()->SetTranslate({-97, 17, -50});
149 }
150
151 // 映画ぽっくする
152 ShowBlackBar();
153
154 // ボスにイントロアニメションが終わったら雑魚エネミーを湧く
155 if(_boss.lock()->IsPlayedTaunt()) {
156 for(auto& mob: _mob) {
157 if(!mob.lock()->GetIsReady()) {
158 mob.lock()->SetToSpawnState();
159 }
160 }
161 _mobSpawnTimer -= GetDeltaTime60();
162 if(_mobSpawnTimer <= 0) {
163 _isMobSpawned = true;
164 }
165 _scenarioCamera.lock()->SetPositionAndTarget(CUT_SCENE_POS_MOB, CUT_SCENE_TARGET_MOB);
166 }
167
168 // 雑魚エネミー湧いたらゲーム始まる
169 if(_isMobSpawned) {
170 _slideBlackBar = true;
171 _playerCamera.lock()->SetTranslate({-97, 17, -50});
172 _cutSceneTimer -= GetDeltaTime60();
173 _cutSceneTimer = std::max(0.0f, _cutSceneTimer);
174 t = abs(1 - (_cutSceneTimer / START_CUT_SCENE_TIME));
175 newCamPos = lerp(CUT_SCENE_POS_MOB, _playerCamera.lock()->GetTranslate(), t);
176 newCamTarget = lerp(CUT_SCENE_TARGET_MOB, _player.lock()->GetTranslate(), t);
177 newFOV = lerp(float1(FOV_INTRO), FOV_ORG, t);
178 _scenarioCamera.lock()->SetPositionAndTarget(newCamPos, newCamTarget);
179
180 _scenarioCamera.lock()->SetPerspective(newFOV);
181 }
182
183 // イントロ演出をスキップ
184 if(IsKeyDown(KEY_INPUT_RETURN) || IsMouseDown(MOUSE_INPUT_1) || IsKeyDown(KEY_INPUT_SPACE) ||
185 IsPadOn(PAD_ID::PAD_L_PUSH) || IsPadOn(PAD_ID::PAD_B)) {
186 _fadeTimer = 0;
187 _alpha = 0;
188 _cutSceneTimer = 0;
189 _scenarioCamera.lock()->SetPerspective(FOV_ORG);
190 }
191
192 // 演出終わったらUIを表示する
193 if(_cutSceneTimer <= 0) {
194 _playerCamera.lock()->SetCurrentCamera();
195 scene_state = Scene::SceneState::GAME;
196 _player.lock()->SetSceneState(scene_state);
197 _player.lock()->SetHideHP(false);
198 _boss.lock()->SetSceneState(scene_state);
199 _boss.lock()->SetHideHP(false);
200 for(auto& mob: _mob) {
201 mob.lock()->SetSceneState(scene_state);
202 mob.lock()->SetHideHP(false);
203 }
204 }
205 break;
206 case Scene::SceneState::GAME:
207 // もしチュートリアル表示してないならチュートリアル表示する
208 if(!_isShowedTutorial) {
209 Scene::Pause();
210 Scene::ShowTutorial();
211 _isShowedTutorial = true;
212 }
213
214 _second -= GetDeltaTime();
215 _showBlackBar = false;
216 if(_second <= 0) {
217 if(_minute <= 0) {
218 _isLose = true;
219 _second = 0.0f;
220 } else {
221 _second = 59.99f;
222 }
223 _minute--;
224 _minute = std::max(0, _minute);
225 }
226
227 // 必殺技演出中にUIを隠す
228 if(_player.lock()->IsPlayingEXSkill()) {
229 _player.lock()->SetHideHP(true);
230 _boss.lock()->SetHideHP(true);
231 for(auto& mob: _mob) {
232 mob.lock()->SetHideHP(true);
233 }
234 } else {
235 _player.lock()->SetHideHP(false);
236 _boss.lock()->SetHideHP(false);
237 for(auto& mob: _mob) {
238 mob.lock()->SetHideHP(false);
239 }
240 }
241
242 // プレイヤーかボスどちらか倒されたらフィニッシュ演出開始します
243 if(_boss.lock()->IsDead() || _player.lock()->IsDead() || _isLose) {
244 _showBlackBar = true;
245 _slideBlackBar = false;
246 if(_cutSceneTimer == START_CUT_SCENE_TIME) {
247 _player.lock()->SlowMotion();
248 _boss.lock()->SlowMotion();
249 }
250 _cutSceneTimer -= GetDeltaTime60();
251 _cutSceneTimer = std::max(0.0f, _cutSceneTimer);
252 _scenarioCamera.lock()->SetCurrentCamera();
253 if(_cutSceneTimer > 120.0f) {
254 _scenarioCamera.lock()->SetPositionAndTarget(_player.lock()->GetTranslate() + float3{20, 15, 20},
255 _player.lock()->GetTranslate() + float3{0, 5, 0});
256 } else {
257 _scenarioCamera.lock()->SetPositionAndTarget(_boss.lock()->GetTranslate() + float3{-40, 15, -40},
258 _boss.lock()->GetTranslate() + float3{0, 5, 0});
259 }
260
261 _player.lock()->SetHideHP(true);
262 _boss.lock()->SetHideHP(true);
263 for(auto& mob: _mob) {
264 mob.lock()->SetHideHP(true);
265 }
266 } else {
267 _cutSceneTimer = START_CUT_SCENE_TIME;
268 }
269
270 if(_cutSceneTimer <= 0 && FadeOut()) {
271 scene_state = Scene::SceneState::TRANS_OUT;
272 _player.lock()->SetSceneState(scene_state);
273 _boss.lock()->SetSceneState(scene_state);
274 _player.lock()->SetTranslate(PLAYER_SPAWN_POS);
275 _boss.lock()->SetTranslate(BOSS_SPAWN_POS);
276 _boss.lock()->SetRotationAxisXYZ({0, 90, 0});
277 _player.lock()->EndSlowMotion();
278 _boss.lock()->EndSlowMotion();
279 }
280 break;
281 case Scene::SceneState::TRANS_OUT:
282 _showBlackBar = false;
283 Scene::SetCanPause(false);
284 _endingTimer -= GetDeltaTime60();
285
286 StopSoundMem(_BGM);
287 StopSoundMem(_introBGM);
288 _boss.lock()->PlayDead();
289 _player.lock()->PlayDead();
290
291 _scenarioCamera.lock()->SetCurrentCamera();
292 if(_boss.lock()->IsDead()) {
293 _scenarioCamera.lock()->SetPositionAndTarget(BOSS_DEATH_CAM, _boss.lock()->GetTranslate() + float3{0, 15, 0});
294 _showImage = _clearImage.lock();
295 } else {
296 _scenarioCamera.lock()->SetPositionAndTarget(PLAYER_DEATH_CAM, _player.lock()->GetTranslate() + float3{0, 10, 0});
297 _showImage = _failImage.lock();
298 }
299 break;
300 }
301}

303void Stage01::LateDraw() {
304 int screen_width, screen_height;
305 GetScreenState(&screen_width, &screen_height, NULL);
306
307 float blackbarY = 1.0f;
308 if(_slideBlackBar) {
309 blackbarY = (_cutSceneTimer / START_CUT_SCENE_TIME);
310 }
311
312 if(_showBlackBar) {
313 DrawBoxAA(0, 0, screen_width, screen_height * 0.15f * blackbarY, 0u, TRUE);
314 DrawBoxAA(0, screen_height - screen_height * 0.15f * blackbarY, screen_width, screen_height, 0u, TRUE);
315 }
316
317 SetDrawBlendMode(DX_BLENDMODE_ALPHA, (int)_alpha);
318 DrawBox(0, 0, screen_width, screen_height, 0u, TRUE);
319 SetDrawBlendMode(DX_BLENDMODE_NOBLEND, NULL);
320 int timerColor = 0xffffff;
321 if(_minute <= 0) {
322 timerColor = 0xff0000;
323 }
324
325 switch(scene_state) {
326 case Scene::SceneState::TRANS_IN:
327 break;
328 case Scene::SceneState::GAME:
329 if(!(_boss.lock()->IsDead() || _player.lock()->IsDead() || _player.lock()->IsPlayingEXSkill())) {
330 DrawFormatStringToHandle((int)((screen_width * 0.9f) - (_stringWidth * 0.5f)), (int)(screen_height * 0.1),
331 timerColor, _timerFontHandle, "%02i:%06.3f", _minute, _second);
332 }
333 break;
334 case Scene::SceneState::TRANS_OUT:
335 if(FadeIn()) {
336 if(ShowMessage()) {
337 DrawStringToHandle((int)((screen_width * 0.5f) - (_stringWidth * 0.5f)), (int)(screen_height * 0.8),
338 "Press any key back to Title", 0xffee42, _fontHandle, 0xffaf3f);
339 if(IsKeyDown(KEY_INPUT_RETURN) || IsMouseDown(MOUSE_INPUT_1) || IsKeyDown(KEY_INPUT_SPACE) ||
340 IsKeyDown(KEY_INPUT_ESCAPE) || IsPadOn(PAD_ID::PAD_R_PUSH) || IsPadOn(PAD_ID::PAD_B) || _endingTimer <= 0) {
341 Scene::Change(Scene::GetScene<GameTitleScene>());
342 }
343 } else {
344 if(IsKeyDown(KEY_INPUT_RETURN) || IsMouseDown(MOUSE_INPUT_1) || IsKeyDown(KEY_INPUT_SPACE) ||
345 IsKeyDown(KEY_INPUT_ESCAPE) || IsPadOn(PAD_ID::PAD_R_PUSH) || IsPadOn(PAD_ID::PAD_B)) {
346 _shrinkTimer = SHRINK_TIME;
347 }
348 }
349 _showImage.lock()->SetPosition(
350 (screen_width * 0.2f), (screen_height * (0.0f + (0.4f * (_shrinkTimer / SHRINK_TIME)))), (screen_width * 0.8f),
351 (screen_height * (1.0f - (0.4f * (_shrinkTimer / SHRINK_TIME)))));
352 _showImage.lock()->DrawTexture();
353 }
354 break;
355 }
356}

373bool Stage01::FadeIn() {
374 if(_fadeTimer > 0) {
375 _fadeTimer -= GetDeltaTime60();
376 _alpha = abs(_fadeTimer / FADE_TIME) * 255;
377 }
378
379 return _fadeTimer <= 0;
380}
381
382bool Stage01::FadeOut() {
383 if(_fadeTimer < FADE_TIME) {
384 _fadeTimer += GetDeltaTime60();
385 _alpha = abs(_fadeTimer / FADE_TIME) * 255;
386 }
387
388 return _fadeTimer >= FADE_TIME;
389}
