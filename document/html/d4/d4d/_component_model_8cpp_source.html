13static int Combo(const std::string& caption, std::string& current_item, const std::vector<std::string_view>& items) {
14 int select_index = -1;
15 for(int i = 0; i < items.size(); i++) {
16 auto& item = items[i];
17 if(current_item == item) {
18 select_index = i;
19 break;
20 }
21 }
22
23 if(ImGui::BeginCombo(caption.c_str(), current_item.c_str())) {
24 for(int i = 0; i < items.size(); i++) {
25 auto& item = items[i];
26
27 bool is_selected = (current_item == item);
28 if(ImGui::Selectable(item.data(), is_selected)) {
29 current_item = item;
30 select_index = i;
31 }
32 if(is_selected)
33 ImGui::SetItemDefaultFocus();
34 }
35 ImGui::EndCombo();
36 }
37
38 return select_index;
39}

67void ComponentModel::Load(std::string_view path) {
68 model_status_.off(ModelBit::ErrorFileNotFound);
69 model_status_.on(ModelBit::SetFileAlready);

71 path_ = path;
72
73 // ファイルが存在しているか?

92 nodes_name_.clear();

103//! @brief モデル更新
104//! @param delta 1フレームの秒数
105void ComponentModel::Update() {
106 float delta = GetDeltaTime();
107
108 // モデルが存在しているならばTransform設定を行う
109 if(IsValid()) {
110 // アニメーションがあり再生している?
111 if(animation_ && animation_->isValid() && animation_->isPlaying()) {
112 animation_->update(delta);
113 animation_time_ += delta;
114
115 // 進めた結果終了した場合はOldとして確保
116 if(!animation_->isPlaying()) {
117 old_animation_name_ = current_animation_name_;
118 }
119 }
120
121 auto mat = model_transform_;
122 auto trns = GetOwner()->GetComponent<ComponentTransform>();
123 if(trns) {
124 mat = mul(mat, trns->GetWorldMatrix());
125 }
126
127 if(model_) {
128 // ワールド行列を設定
129 model_->setWorldMatrix(mat);
130
131 // モデル更新
132 model_->update(delta);
133 }
134 }

137//! @brief モデル描画
138void ComponentModel::Draw() {
139 if(!model_status_.is(ModelBit::Initialized))
140 return;
141
142 // Drawはここで抑えておく
143 if(GetStatus(Component::StatusBit::NoDraw))
144 return;
145
146 if(model_ == nullptr)
147 return;
148
149 // ワールド行列を設定(コリジョン移動分)
150 model_->setWorldMatrix(GetWorldMatrix());
151
152 // シェーダーを利用するかどうかを設定
153 model_->useShader(UseShader());
154
155 std::shared_ptr<Texture> empty = nullptr;
156
157 auto drawFrame = [this, &empty](int i) {
158 if(materials_.find(i) != materials_.end()) {
159 // マテリアルが存在した
160 model_->overrideTexture(Model::TextureType::Diffuse, materials_[i].diffuse_);
161 model_->overrideTexture(Model::TextureType::Normal, materials_[i].normal_);
162 model_->overrideTexture(Model::TextureType::Roughness, materials_[i].roughness_);
163 model_->overrideTexture(Model::TextureType::Metalness, materials_[i].metalness_);
164 model_->overrideTexture(Model::TextureType::AO, materials_[i].AO_);
165 model_->overrideTexture(Model::TextureType::Specular, materials_[i].specular_);
166 } else {
167 // 存在せず
168 model_->overrideTexture(Model::TextureType::Diffuse, empty);
169 model_->overrideTexture(Model::TextureType::Normal, empty);
170 model_->overrideTexture(Model::TextureType::Roughness, empty);
171 model_->overrideTexture(Model::TextureType::Metalness, empty);
172 model_->overrideTexture(Model::TextureType::AO, empty);
173 model_->overrideTexture(Model::TextureType::Specular, empty);
174 }
175 model_->renderByFrame(i, overrided_shader_vs_, overrided_shader_ps_);
176 };
177
178 // モデル描画
179 if(draw_meshes_.size() > 0 && draw_meshes_[0] == -1) {
180 if(materials_.find(-1) != materials_.end()) {
181 model_->overrideTexture(Model::TextureType::Diffuse, materials_[-1].diffuse_);
182 model_->overrideTexture(Model::TextureType::Normal, materials_[-1].normal_);
183 model_->overrideTexture(Model::TextureType::Roughness, materials_[-1].roughness_);
184 model_->overrideTexture(Model::TextureType::Metalness, materials_[-1].metalness_);
185 model_->overrideTexture(Model::TextureType::AO, materials_[-1].AO_);
186 model_->overrideTexture(Model::TextureType::Specular, materials_[-1].specular_);
187 model_->render();
188 } else {
189 for(int i = 0; i < model_->frameCount(); ++i) {
190 drawFrame(i);
191 }
192 }
193 return;
194 }
195
196 for(int i: draw_meshes_) {
197 drawFrame(i);
198 }
199}

673//! @brief ワールドMatrixの取得
674//! @return 他のコンポーネントも含めた位置
675const matrix ComponentModel::GetWorldMatrix() const {
676 matrix mat = matrix::identity();
677
678 // AttachedOtherModelの場合はそのコンポーネントもマージする
679 if(IsAttachedOtherModel()) {
680 mat = GetAttachMatrix();
681
682 // スケールを自前のものに戻す
683 if(!use_model_node_scale_) {
684 matrix parent_mat(float4{normalize(mat.axisX()), 0}, float4{normalize(mat.axisY()), 0},
685 float4{normalize(mat.axisZ()), 0}, float4{mat.translate(), 1});
686 mat = parent_mat;
687 }
688 }
689
690 mat = mul(GetMatrix(), mat);
