64 [[maybe_unused]] const JPH::CollideShapeResult& collision_result) override {

70
71 //broad_phase_layer_interfaceでやるべきことをここで行う todo broad_phase_layer_interfaceの変更
72 if(!(ComponentPhysics::GetHitCollision(ct1) & (u16)ct2) || !(ComponentPhysics::GetHitCollision(ct2) & (u16)ct1)) {
73 // ヒットしない
74 return JPH::ValidateResult::RejectAllContactsForThisBodyPair;
75 }
76
77 ComponentPhysics::HitInfo info;
78 info.hit_ = true;
79 info.collision_ = pd1;
80 info.hit_collision_ = pd2;
81 pd1->OnHit(info);
82

84 info.collision_ = pd2;
85 info.hit_collision_ = pd1;
86 pd2->OnHit(info);
87
