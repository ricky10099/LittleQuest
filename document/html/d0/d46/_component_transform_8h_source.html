240 //! @brief ベクトルの方向へ向ける
241 //! @param vec 向きたいベクトル
242 //! @param limit 向ける限界角度
243 //! @param up_change Y軸も変更するか?
244 //! @return 自分のSharedPtr
245 auto SetRotationToVectorWithLimit(float3 vec, float limit, float3 up = {0, 1, 0}, bool up_change = false) {
246 float delta = GetDeltaTime60();
247 limit = limit * delta;
248
249 // 今の姿勢マトリクス
250 auto now_mat = GetMatrix();
251
252 // 向きたい方向のマトリクス
253 auto mat = HelperLib::Math::CreateMatrixByFrontVector(-vec, up, up_change);
254
255 // 現在の姿勢
256 float matrixTranslation[3], matrixRotation[3], matrixScale[3];
257 // 向きたい姿勢
258 float matrixTranslationVec[3], matrixRotationVec[3], matrixScaleVec[3];
259
260 DecomposeMatrixToComponents(GetMatrixFloat(), matrixTranslation, matrixRotation, matrixScale);
261
262 DecomposeMatrixToComponents(mat.f32_128_0, matrixTranslationVec, matrixRotationVec, matrixScaleVec);
263
264 float rot_x = matrixRotationVec[0] - matrixRotation[0];
265 if(rot_x > 180)
266 rot_x -= 360;
267 if(rot_x < -180)
268 rot_x += 360;
269 float rot_y = matrixRotationVec[1] - matrixRotation[1];
270 if(rot_y > 180)
271 rot_y -= 360;
272 if(rot_y < -180)
273 rot_y += 360;
274 float rot_z = matrixRotationVec[2] - matrixRotation[2];
275 if(rot_z > 180)
276 rot_z -= 360;
277 if(rot_z < -180)
278 rot_z += 360;
279
280 if(rot_x > 0 && rot_x > limit)
281 rot_x = limit;
282 if(rot_x < 0 && rot_x < -limit)
283 rot_x = -limit;
284
285 if(rot_y > 0 && rot_y > limit)
286 rot_y = limit;
287 if(rot_y < 0 && rot_y < -limit)
288 rot_y = -limit;
289
290 if(rot_z > 0 && rot_z > limit)
291 rot_z = limit;
292 if(rot_z < 0 && rot_z < -limit)
293 rot_z = -limit;
294
295 matrixRotationVec[0] = rot_x + matrixRotation[0];
296 matrixRotationVec[1] = rot_y + matrixRotation[1];
297 matrixRotationVec[2] = rot_z + matrixRotation[2];
298 RecomposeMatrixFromComponents(matrixTranslation, matrixRotationVec, matrixScale, GetMatrixFloat());
299 return SharedThis();
300 }
