20struct Variant {
21 //! デフォルトコンストラクタ
22 Variant() = default;
23 //! デストラクタ
24 ~Variant() {
25 if(handle_ != -1) {
26 DxLib::DeleteShader(handle_);
27 }
28 }

30 //! @param [in] source_path HLSLソースコードのファイルパス
31 //! @param [in] dxlib_shader_type [DxLib] シェーダーの種類(DX_SHADERTYPE_VERTEXなど)
32 bool compile(const std::wstring& source_path, u32 index, int dxlib_shader_type) {
33 //---------------------------------------------------------------------
34 // ファイルからソースファイルを読み込み
35 //---------------------------------------------------------------------
36 std::vector<std::byte> source_code;
37 {
38 // ファイルから読み込み
39 std::ifstream file(
40 source_path.c_str(),
41 std::ios::in | std::ios::binary | std::ios::ate); // ateを指定すると最初からファイルポインタが末尾に移動
42 if(!file.is_open()) {
43 return false;
44 }
45 auto size = file.tellg();
46 source_code.resize(static_cast<size_t>(size));
47
48 file.seekg(0, std::ios::beg);
49 file.read(reinterpret_cast<char*>(source_code.data()), size);
50 file.close();
51 }
52
53 //--------------------------------------------------------------
54 // シェーダーをコンパイル
55 //--------------------------------------------------------------
56 u32 compile_flags = D3DCOMPILE_OPTIMIZATION_LEVEL3;
57 compile_flags |= D3DCOMPILE_PACK_MATRIX_ROW_MAJOR;
58
59#if defined(_DEBUG)
60 // GraphicDebuggingツール用にシェーダーデバッグ向け設定
61 compile_flags |= D3DCOMPILE_DEBUG | D3DCOMPILE_SKIP_OPTIMIZATION;
62#endif
63
64 // #define SHADER_VARIANT index の形式でプリプロセッサ定義
65 auto number_string = std::to_string(index);
66
67 auto dxlib_version = std::format("{:#x}", DXLIB_VERSION);
68
69 // シェーダーモデル名
70 static const char* target_names[]{
71 "vs_5_0", // DX_SHADERTYPE_VERTEX // 頂点シェーダー
72 "ps_5_0", // DX_SHADERTYPE_PIXEL // ピクセルシェーダー
73 "gs_5_0", // DX_SHADERTYPE_GEOMETRY // ジオメトリシェーダー
74 "cs_5_0", // DX_SHADERTYPE_COMPUTE // コンピュートシェーダー
75 "ds_5_0", // DX_SHADERTYPE_DOMAIN // ドメインシェーダー
76 "hs_5_0", // DX_SHADERTYPE_HULL // ハルシェーダー
77 };
78
79 // シェーダー内で参照できる #define マクロ定義
80 const D3D_SHADER_MACRO defines[]{
81 {"SHADER_VARIANT", number_string.c_str()},
82 { "DXLIB_VERSION", dxlib_version.c_str()},
83 { nullptr, nullptr},
84 };
85
86 Microsoft::WRL::ComPtr<ID3DBlob> byte_code = nullptr;
87 Microsoft::WRL::ComPtr<ID3DBlob> errors;
88
89 auto hr = D3DCompile(source_code.data(), // [in] ソースコードのメモリ上のアドレス
90 source_code.size(), // [in] ソースコードサイズ
91 convertTo(source_path).c_str(), // [in] ソースコードのファイルパス(使用しない場合はnullptr)
92 defines, // [in] プリプロセッサマクロ定義
93 D3D_COMPILE_STANDARD_FILE_INCLUDE, // [in] カスタムインクルード処理
94 "main", // [in] 関数名
95 target_names[dxlib_shader_type], // [in] シェーダーモデル名
96 compile_flags, // [in] コンパイラフラグ (D3DCOMPILE_xxxx)
97 0, // [in] コンパイラフラグ2 (D3DCOMPILE_FLAGS2_xxxx)
98 &byte_code, // [out] コンパイルされたバイトコード
99 &errors); // [out] エラーメッセージ
100
101 // エラー警告出力
102 if(errors != nullptr) {
103 // 「出力」ウィンドウに表示
104 OutputDebugStringA("--------------------\n");
105 OutputDebugStringA((char*)errors->GetBufferPointer());
106 OutputDebugStringA("--------------------\n");
107
108 // メッセージボックス
109 auto file_name = convertTo(source_path);
110 MessageBox(DxLib::GetMainWindowHandle(), static_cast<char*>(errors->GetBufferPointer()), file_name.c_str(),
111 MB_ICONWARNING | MB_OK);
112 }
113
114 if(FAILED(hr)) {
115 return false;
116 }
117
118 //------------------------------------------------------
119 // [DxLib] シェーダーを作成
120 //------------------------------------------------------
121 const void* shader_bytecode = byte_code->GetBufferPointer(); // シェーダーバイトコードの先頭アドレス
122 auto shader_size = static_cast<int>(byte_code->GetBufferSize()); // シェーダーバイトコードのサイズ
123
124 // シェーダーバイトコードを保存
125 shader_bytecode_.resize(shader_size);
126 memcpy(shader_bytecode_.data(), shader_bytecode, shader_size);
127
128 int handle = -1;
129
130 switch(dxlib_shader_type) {
131 case DX_SHADERTYPE_VERTEX: // 頂点シェーダー
132 handle = LoadVertexShaderFromMem(shader_bytecode, shader_size);
133 break;
134 case DX_SHADERTYPE_PIXEL: // ピクセルシェーダー
135 handle = LoadPixelShaderFromMem(shader_bytecode, shader_size);
136 break;
137 case DX_SHADERTYPE_GEOMETRY: // ジオメトリシェーダー
138 handle = LoadGeometryShaderFromMem(shader_bytecode, shader_size);
139 break;
140 default:
141 break;
142 }
143
144 //------------------------------------------------------
145 // 結果を出力
146 //------------------------------------------------------
147 handle_ = handle;
148
149 return true;
150 }
151
152 int handle_ = -1; //!< [DxLib] シェーダーハンドル
153 std::vector<std::byte> shader_bytecode_; //!< シェーダーバイトコード
154};

156//----------------------------------------------------------
157//! シェーダー実体
158//----------------------------------------------------------
159struct Shader::Impl {
160 std::wstring path_; //!< ファイルパス
161 int type_ = -1; //!< [DxLib] シェーダーの種類(DX_SHADERTYPE_VERTEXなど)
162 std::vector<Variant> variant_; //!< バリエーションごとの定義
163
164 //! 現在アクティブなシェーダーリスト
165 static inline std::unordered_map<std::wstring, std::shared_ptr<Shader::Impl>> shaders_;
166
167 public:
168 //! @param [in] path HLSLソースコードのファイルパス
169 //! @param [in] dxlib_shader_type [DxLib] シェーダーの種類(DX_SHADERTYPE_VERTEXなど)
170 Impl(std::wstring_view path, int dxlib_shader_type, u32 variant_count) {
171 path_ = path;
172 type_ = dxlib_shader_type;
173
174 // 最低限1つのバリエーションを持つ
175 if(variant_count == 0)
176 variant_count = 1;
177
178 variant_.resize(variant_count);
179
180 compile();
181 }
182
183 bool compile() {
184 // シェーダーソースの拡張子のパスを作成
185 std::wstring source_path = path_ + L".fx";
186
187 //----------------------------------------------------------
188 // シェーダーバリエーションコンパイル
189 //----------------------------------------------------------
190 for(u32 i = 0; i < variant_.size(); i++) {
191 // シェーダーをコンパイル
192 Variant new_variant;
193 bool result = new_variant.compile(source_path, i, type_);
194
195 // 失敗したらその先の更新をスキップ終了
196 if(!result) {
197 return false;
198 }
199
200 // 作成が成功したら旧シェーダーを解放して新しい結果に置換
201 // 代入ではなくstd::swapで行う理由は、交換されたあとのローカル変数が自動解放されるのを利用して
202 // 旧シェーダーをデストラクタで解放させることで簡潔に記述することができる。
203 std::swap(variant_[i], new_variant);
204 }
205
206 return true;
207 }
208};

220Shader::Shader(std::string_view source_path, u32 dxlib_shader_type, u32 variant_count) {
221 assert(dxlib_shader_type <= DX_SHADERTYPE_HULL);
222
223 //----------------------------------------------------------
224 // ファイルパス文字列を正規化
225 //----------------------------------------------------------
226 std::filesystem::path path(source_path);
227
228 // 拡張子を除去したファイルパスを取得
229 auto shader_path = path.parent_path().wstring() + L"/" + path.stem().wstring();
230
231 auto normalize_path = [](wchar_t c) {
232 // 小文字に統一することで大文字小文字の差異をなくす
233 c = ::towlower(c);
234
235 // ファイルパス記号 '\\' を '/' に統一
236 if(c == '\\') {
237 c = '/';
238 }
239 return c;
240 };
241
242 std::transform(shader_path.begin(), shader_path.end(), shader_path.begin(), normalize_path);
243
244 //-----------------------------------------------------------------------
245 // 初回実行時にファイル監視を開始する
246 //-----------------------------------------------------------------------
247 // ファイル変更時のコールバック関数
248 auto file_modified_callback = [=](const wchar_t* full_path) {
249 // ファイルパスの要素を分解
250 std::filesystem::path path(full_path);
251 auto file_path = path.parent_path().wstring() + L"/" + path.stem().wstring();
252
253 // シェーダーソースコードだった場合
254 if(path.extension() == L".fx") {
255 // ファイルパスを正規化
256 std::transform(file_path.begin(), file_path.end(), file_path.begin(), normalize_path);
257
258 // 読み込み済のシェーダーパスと照合して一致したらホットリード対象
259 auto it = Shader::Impl::shaders_.find(file_path);
260
261 if(it != std::end(Shader::Impl::shaders_)) {
262 auto& shader_impl = it->second;
263 shader_impl->compile(); // 対象のシェーダーをコンパイル
264 }
265 }
266 };
267
268 // ファイル監視開始
269 std::call_once(once_initialize_, [&]() {
270 file_watcher_.initialize(L".", file_modified_callback);
271 });
272
273 //----------------------------------------------------------
274 // 初期設定
275 //----------------------------------------------------------
276 // シェーダーリストを検索して既存のシェーダーの場合は再利用
277 auto it = Shader::Impl::shaders_.find(shader_path);
278
279 if(it == std::end(Shader::Impl::shaders_)) { // 見つからなかった場合は新規作成
280 auto p = std::make_shared<Shader::Impl>(shader_path, dxlib_shader_type, variant_count);
281
282 // シェーダーリストに登録
283 Shader::Impl::shaders_[shader_path] = p;
284
285 impl_ = std::move(p);
286 } else { // 見つかった場合は再利用
287 impl_ = it->second;
288 }
289}
