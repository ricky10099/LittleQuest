142void MobEnemy::Patrol() {
143 auto pos = GetTranslate();
144 _movement = _goal - pos;
145 _movement.y = 0;
146 float moveValue = GetDistance(_movement);
147
148 if(moveValue <= TRIGGER_POINT_DISTANCE) {
149 _patrolIndex++;
150 _patrolIndex %= _patrolPoint.size();
151 _goal = _patrolPoint[_patrolIndex];
152 _waitFor = PATROL_WAIT_TIME;
153 ChangeState(MobEnemyState::WAIT);
154 return;

156
157 if(moveValue > TRIGGER_POINT_DISTANCE) {
158 _movement = normalize(_movement);
159
160 float x = -_movement.x;
161 float z = -_movement.z;
162 float theta = atan2(x, z) * RadToDeg;
163 SetRotationAxisXYZ({0, theta, 0});

178void MobEnemy::ChasePlayer() {

215void MobEnemy::GameAction() {
216 if(_isDead) {
217 return;
218 }
219
220 switch(_state) {
221 case MobEnemyState::GET_HIT:
222 if(!_model.lock()->IsPlaying()) {
223 ChangeState(MobEnemyState::CHASE);
224 }
225 break;
226 case MobEnemyState::GIVE_UP:
227 BackToInitialPosition();
228 break;
229 case MobEnemyState::CHASE:
230 FindPlayer();
231 ChasePlayer();
232 break;
233 case MobEnemyState::ATTACK:
234 Attack();
235 break;
236 case MobEnemyState::WAIT:
237 Wait();
238 break;
239 case MobEnemyState::PATROL:
240 Patrol();
241 FindPlayer();
242 break;
243 case MobEnemyState::IDLE:
244 Idle();
245 FindPlayer();
246 break;
247 }
248
249 _movement *= _baseSpeed * _speedFactor * GetDeltaTime60();
250 AddTranslate(_movement);
251}
