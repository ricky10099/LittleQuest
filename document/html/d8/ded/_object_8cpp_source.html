34//! @brief 名前から、ユニークな名前を設定します
35//! @param name つけたい名前
36//! @return 通常はそのまま返しますが、同じ名前がある場合は違う名前に変更されます
37std::string Object::setUniqueName(const std::string& name) {
38 if(!_obj_names_count || !_obj_names_id) {
39 _obj_names_count = std::make_unique<NamesID>();
40 _obj_names_id = std::make_unique<NamesID>();
41 _obj_names_count_adrs = &_obj_names_count;
42 }
43 NamesID& obj_names_count = *_obj_names_count;
44 NamesID& obj_names_id = *_obj_names_id;
45
46 // 今の名前のものがほかにいないか?
47 auto xname = std::string(GetNameDefault());
48 auto xitr = obj_names_count.find(xname);
49 if(xitr != obj_names_count.end()) {
50 s64 count = --obj_names_count[xname];
51 if(count <= 0) {
52 obj_names_count.erase(xname);
53 obj_names_id.erase(xname);
54 }
55 }
56
57 auto itr = obj_names_id.find(name);
58
59 if(itr != obj_names_id.end()) {
60 // 登録済なら連番で別名をつける
61 auto str = name + "_" + std::to_string(obj_names_id[name]);
62 obj_names_id[name]++;
63 obj_names_count[name]++;
64 return str;
65 }
66 // 登録済の名前がなかった場合は新規登録
67 obj_names_id[name]++;
68 obj_names_count[name]++;
69 return name;
70}

321void Object::ModifyComponents() {
322 auto& components = GetComponents();
323 for(int i = (int)components.size() - 1; i >= 0; --i) {
324 auto& c = components[i];
325 if(c->status_.is(Component::StatusBit::Exited)) {
326 auto comp = components.begin() + i;
327
328 ComponentWeakPtr weak_comp = *comp;
329 components.erase(comp); //解放処理(自動delete)
330
331 if(weak_comp.lock() != nullptr) {
332 // どこかに残っているので一旦確保
333 leak_components_.push_back(weak_comp);
334 }
335 }
336 }
