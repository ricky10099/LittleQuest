28 //----------------------------------------------------------

37 if(child) {
38 siblings_ = child;

40 parent_type->child_ = this;
41 }
42}

100//! 名前を指定して指定基底クラス型でnewする
101//---------------------------------------------------------------------------
102void* CreateInstanceFromName(std::string_view class_name, const TypeInfo& base_type) {
103 const TypeInfo* p = base_type.child();
104 bool returnFromTraverse = false;
105 const TypeInfo* next = nullptr;
106
107 //----------------------------------------------------------
108 // 継承ツリー構造を探索
109 // スタック再帰を使わない高速なツリー探索 (stackless tree traversal)
110 //----------------------------------------------------------
111 while(p != &base_type) {
112 if(!returnFromTraverse) {
113 // 名前チェックして一致したら作成
114 if(p->className() == class_name) {
115 return p->createInstance();
116 }
117 }
118
119 if(p->child() && !returnFromTraverse) {
120 // 子がある場合は子を先に調べる。(子から探索で戻ってきた場合は除外)
121 next = p->child();
122 returnFromTraverse = false;
123 } else if(p->siblings()) {
124 // 兄弟がいる場合は兄弟を調べる
125 next = p->siblings();
126 returnFromTraverse = false;
127 } else {
128 // 親へ戻る。
129 next = p->parent();
130 returnFromTraverse = true;
131 }
132 p = next;
133 }
134
135 return nullptr;
136}
