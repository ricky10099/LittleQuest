685//! @brief オブジェクト取得
686//! @tparam T 取得したいオブジェクトクラス
687//! @return 取得オブジェクト
688template<class T>
689std::shared_ptr<T> Scene::Base::GetObjectPtr(const std::string_view name) {
690 if(name.empty()) {
691 for(auto& obj: objects_) {
692 auto cast = std::dynamic_pointer_cast<T>(obj);
693
694 if(cast)
695 return cast;
696 }
697 } else {
698 for(auto& obj: objects_) {
699 if(name.compare(obj->GetNameDefault()) == 0) {
700 auto cast = std::dynamic_pointer_cast<T>(obj);
701
702 if(cast)
703 return cast;
704 }
705 }
706 for(auto& obj: objects_) {
707 if(name.compare(obj->GetName()) == 0) {
708 auto cast = std::dynamic_pointer_cast<T>(obj);
709
710 if(cast)
711 return cast;
712 }
713 }
714
715 // 作成前Objectも検査する
716 for(auto& obj: pre_objects_) {
717 if(name.compare(obj->GetNameDefault()) == 0) {
718 auto cast = std::dynamic_pointer_cast<T>(obj);
719
720 if(cast)
721 return cast;
722 }
723 }
724 for(auto& obj: pre_objects_) {
725 if(name.compare(obj->GetName()) == 0) {
726 auto cast = std::dynamic_pointer_cast<T>(obj);
727
728 if(cast)
729 return cast;
730 }
731 }
732 }
733
734 return nullptr;
735}
