122//---------------------------------------------------------------------------
123//! クラス内ヘッダー宣言
124//! publicで BP_CLASS_TYPE(クラス名, u8"任意の解説文字列") で宣言します

133#define BP_CLASS_DECL(CLASS, ...) \

137 /*! 型情報 */ \

139 \

142 return &Type; \
143 } \
144 \

146 static void* createInstance() { \
147 return CreateInstance<CLASS>()(); \
